# SimpleChat

A real-time messaging application built with Django.

## Features

- User registration
- Login and logout
- Private conversations
- Group conversations
- Real-time messaging
- PostgreSQL database
- Redis channel layer
- Django Channels
- Responsive interface

## Requirements

- Python 3
- Docker Desktop
- VS Code (recommended)

## Installation

1. Install Python 3.
2. Install Docker Desktop.
3. Open the project folder in Terminal.
4. Run:

from django.contrib import admin
from django.contrib.auth import get_user_model
from django.contrib.auth.admin import UserAdmin

from .models import (
    Block,
    Conversation,
    Membership,
    Message,
    Profile,
    Report,
    SignupAttempt,
)


User = get_user_model()

admin.site.site_header = "SimpleChat Admin"
admin.site.site_title = "SimpleChat Admin"
admin.site.index_title = "Moderation and community"


# ---- Users: ban / unban ----

admin.site.unregister(User)


@admin.register(User)
class ChatUserAdmin(UserAdmin):

    list_display = (
        "username", "is_active", "is_staff", "date_joined", "last_login"
    )

    list_filter = ("is_active", "is_staff")

    actions = ["ban_users", "unban_users"]

    @admin.action(description="Ban selected users (they can no longer log in)")
    def ban_users(self, request, queryset):

        count = queryset.exclude(is_superuser=True).update(is_active=False)

        self.message_user(request, f"{count} user(s) banned.")

    @admin.action(description="Unban selected users")
    def unban_users(self, request, queryset):

        count = queryset.update(is_active=True)

        self.message_user(request, f"{count} user(s) unbanned.")


# ---- Reports ----

@admin.register(Report)
class ReportAdmin(admin.ModelAdmin):

    list_display = (
        "id", "status", "reporter", "reported_user",
        "short_body", "reason", "created_at"
    )

    list_filter = ("status",)

    search_fields = (
        "reporter__username", "reported_user__username",
        "message_body", "reason"
    )

    actions = [
        "mark_resolved",
        "ban_reported_users",
        "delete_reported_messages",
    ]

    @admin.display(description="Message")
    def short_body(self, obj):

        return obj.message_body[:60]

    @admin.action(description="Mark selected reports as resolved")
    def mark_resolved(self, request, queryset):

        count = queryset.update(status=Report.STATUS_RESOLVED)

        self.message_user(request, f"{count} report(s) resolved.")

    @admin.action(description="Ban the reported users and resolve")
    def ban_reported_users(self, request, queryset):

        user_ids = list(
            queryset
            .exclude(reported_user__isnull=True)
            .values_list("reported_user_id", flat=True)
        )

        banned = (
            User.objects
            .filter(pk__in=user_ids)
            .exclude(is_superuser=True)
            .update(is_active=False)
        )

        queryset.update(status=Report.STATUS_RESOLVED)

        self.message_user(request, f"{banned} user(s) banned.")

    @admin.action(description="Delete the reported messages and resolve")
    def delete_reported_messages(self, request, queryset):

        message_ids = list(
            queryset
            .exclude(message__isnull=True)
            .values_list("message_id", flat=True)
        )

        Message.objects.filter(pk__in=message_ids).delete()

        queryset.update(status=Report.STATUS_RESOLVED)

        self.message_user(request, "Reported messages deleted.")


# ---- Everything else ----

@admin.register(Message)
class MessageAdmin(admin.ModelAdmin):

    list_display = ("id", "sender", "conversation", "short_body", "created_at")

    search_fields = ("body", "sender__username")

    @admin.display(description="Body")
    def short_body(self, obj):

        return obj.body[:60]


@admin.register(Conversation)
class ConversationAdmin(admin.ModelAdmin):

    list_display = ("id", "title", "is_group", "is_default", "created_at")

"""Preset colors and avatars, plus helpers to describe a user's look."""

COLORS = [
    "#4f46e5", "#7c3aed", "#c026d3", "#db2777",
    "#dc2626", "#ea580c", "#d97706", "#65a30d",
    "#16a34a", "#0d9488", "#0891b2", "#2563eb",
]

AVATARS = [
    "😀", "😎", "🤓", "🥳", "😺", "🐶",
    "🦊", "🐼", "🐸", "🦄", "🐙", "🦁",
    "🐧", "🦉", "🌵", "🌈", "🔥", "⚡",
    "🚀", "👾", "🎮", "🎧", "🍕", "🍩",
]


def avatar_info(user):
    """Color, emoji (or '') and initial for a user. Never raises, and works
    for users who never customized anything."""

    profile = getattr(user, "profile", None)

    color = profile.color if profile and profile.color else ""

    if color not in COLORS:
        color = COLORS[user.pk % len(COLORS)]

    emoji = profile.avatar if profile and profile.avatar in AVATARS else ""

    initial = (user.username[:1] or "?").upper()

    return {"color": color, "emoji": emoji, "initial": initial}


def avatar_payload(user):
    """The same info, shaped for the JSON that goes to browsers."""

    info = avatar_info(user)

    return {
        "sender_color": info["color"],
        "sender_avatar": info["emoji"],
        "sender_initial": info["initial"],
    }

from django import forms

from .models import Conversation, Membership


class GroupForm(forms.Form):

    title = forms.CharField(
        max_length=120,

        widget=forms.TextInput(
            attrs={
                "class": "field",
                "placeholder": "Group name"
            }
        ),

        help_text="Groups are open: anyone on the site can find and join them."
    )

    def __init__(self, *args, user=None, **kwargs):

        super().__init__(*args, **kwargs)

        self.user = user

    def save(self):

        conversation = Conversation.objects.create(
            title=self.cleaned_data["title"].strip(),
            is_group=True,
            created_by=self.user
        )

        Membership.objects.create(
            conversation=conversation,
            user=self.user
        )

        return conversation

from django.contrib.auth import get_user_model
from django.core.management.base import BaseCommand

from messaging.models import Membership
from messaging.utils import get_general_room


class Command(BaseCommand):

    help = (
        "Create the General room (if missing) and add every active user "
        "to it. Safe to run more than once."
    )

    def handle(self, *args, **options):

        room = get_general_room()

        existing = set(
            Membership.objects
            .filter(conversation=room)
            .values_list("user_id", flat=True)
        )

        User = get_user_model()

        new = [
            Membership(conversation=room, user_id=user_id)
            for user_id in User.objects
            .filter(is_active=True)
            .values_list("id", flat=True)
            if user_id not in existing
        ]

        Membership.objects.bulk_create(new)

        self.stdout.write(
            self.style.SUCCESS(
                f"General room ready. Added {len(new)} user(s)."
            )
        )

from django.conf import settings
from django.db import models
from django.utils import timezone


class Conversation(models.Model):

    title = models.CharField(
        max_length=120,
        blank=True
    )

    is_group = models.BooleanField(default=False)

    # The group every new user is added to automatically (the "General" room).
    is_default = models.BooleanField(default=False)

    created_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        related_name="created_conversations"
    )

    created_at = models.DateTimeField(
        auto_now_add=True
    )

    def display_name_for(self, user):

        if self.is_group:
            return self.title or "Group chat"

        other = (
            self.members
            .exclude(user=user)
            .select_related("user")
            .first()
        )

        return other.user.username if other else "Direct chat"

    def __str__(self):

        return self.title or f"Conversation {self.pk}"


class Membership(models.Model):

    conversation = models.ForeignKey(
        Conversation,
        on_delete=models.CASCADE,
        related_name="members"
    )

    user = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.CASCADE,
        related_name="chat_memberships"
    )

    joined_at = models.DateTimeField(
        auto_now_add=True
    )

    last_read_at = models.DateTimeField(
        default=timezone.now
    )

    class Meta:

        constraints = [
            models.UniqueConstraint(
                fields=["conversation", "user"],
                name="unique_conversation_member"
            )
        ]


class Message(models.Model):

    conversation = models.ForeignKey(
        Conversation,
        on_delete=models.CASCADE,
        related_name="messages"

"""Ably helpers: sign browser tokens and publish messages.

The API key (settings.ABLY_API_KEY) stays on the server. Browsers only ever
receive short-lived tokens that allow them to *subscribe* to one room.
"""

import base64
import hashlib
import hmac
import json
import time
from urllib.request import Request, urlopen

from django.conf import settings


TOKEN_TTL_SECONDS = 3600


def enabled():

    return bool(getattr(settings, "ABLY_API_KEY", ""))


def channel_name(conversation_id):

    return f"chat-{conversation_id}"


def _b64url(data):

    return base64.urlsafe_b64encode(data).rstrip(b"=").decode()


def make_subscribe_token(user_id, conversation_id):
    """An Ably JWT that can only subscribe to this one room."""

    key = getattr(settings, "ABLY_API_KEY", "")

    if ":" not in key:
        return None

    key_name, secret = key.split(":", 1)

    now = int(time.time())

    header = {"typ": "JWT", "alg": "HS256", "kid": key_name}

    claims = {
        "iat": now,
        "exp": now + TOKEN_TTL_SECONDS,
        "x-ably-capability": json.dumps(
            {channel_name(conversation_id): ["subscribe"]}
        ),
        "x-ably-clientId": str(user_id),
    }

    signing_input = (
        _b64url(json.dumps(header, separators=(",", ":")).encode())
        + "."
        + _b64url(json.dumps(claims, separators=(",", ":")).encode())
    )

    signature = hmac.new(
        secret.encode(),
        signing_input.encode(),
        hashlib.sha256
    ).digest()

    return signing_input + "." + _b64url(signature)


def publish(conversation_id, payload):
    """Publish a message to the room. Returns True on success."""

    key = getattr(settings, "ABLY_API_KEY", "")

    if not key:
        return False

    body = json.dumps({
        "name": "message",
        "data": json.dumps(payload),
    }).encode()

    auth = base64.b64encode(key.encode()).decode()

    request = Request(
        f"https://rest.ably.io/channels/{channel_name(conversation_id)}/messages",
        data=body,
        method="POST",
        headers={
            "Authorization": f"Basic {auth}",
            "Content-Type": "application/json",
        },
    )

    try:
        with urlopen(request, timeout=4) as response:
            response.read()
        return True
    except Exception:
        return False

import json
from urllib.parse import urlencode
from urllib.request import Request, urlopen

from django.conf import settings
from django.db.models import Q

from .models import Block, Conversation, Membership


def client_ip(request):

    real = request.META.get("HTTP_X_REAL_IP", "").strip()

    if real:
        return real[:64]

    forwarded = request.META.get("HTTP_X_FORWARDED_FOR", "")

    if forwarded:
        return forwarded.split(",")[0].strip()[:64]

    return request.META.get("REMOTE_ADDR", "")[:64]


def verify_turnstile(request):
    """Cloudflare Turnstile check. Skipped when no secret key is configured."""

    secret = getattr(settings, "TURNSTILE_SECRET_KEY", "")

    if not secret:
        return True

    token = request.POST.get("cf-turnstile-response", "")

    if not token:
        return False

    data = urlencode({
        "secret": secret,
        "response": token,
        "remoteip": client_ip(request),
    }).encode()

    try:
        with urlopen(
            Request(
                "https://challenges.cloudflare.com/turnstile/v0/siteverify",
                data=data
            ),
            timeout=5
        ) as response:
            return json.load(response).get("success") is True
    except Exception:
        return False


def blocked_ids_for(user):
    """IDs of people `user` has blocked."""

    return set(
        Block.objects
        .filter(blocker=user)
        .values_list("blocked_id", flat=True)
    )


def is_blocked_between(a_id, b_id):
    """True if either person has blocked the other."""

    return Block.objects.filter(
        Q(blocker_id=a_id, blocked_id=b_id)
        | Q(blocker_id=b_id, blocked_id=a_id)
    ).exists()


def get_general_room():

    room = Conversation.objects.filter(is_default=True).first()

    if room is None:
        room = Conversation.objects.create(
            title="General",
            is_group=True,
            is_default=True
        )

    return room


def join_default_rooms(user):

    Membership.objects.get_or_create(
        conversation=get_general_room(),
        user=user
    )

            messages.error(

            user = form.save()

            SignupAttempt.objects.create(ip=ip)

            join_default_rooms(user)

            login(request, user)

            return redirect("inbox")

    return render(
        request,

        {
            "form": form,
            "turnstile_site_key": settings.TURNSTILE_SITE_KEY,

@login_required
def inbox(request):

    memberships = (
        Membership.objects
        .filter(user=request.user)
        .select_related("conversation")
        .order_by("-conversation__created_at")
    )

    chats = []

    for membership in memberships:

        conversation = membership.conversation

        latest = (
            conversation.messages
            .select_related("sender")
            .last()
        )

        unread = (
            conversation.messages
            .filter(created_at__gt=membership.last_read_at)
            .exclude(sender=request.user)
            .count()
        )

        other_user = None

        if not conversation.is_group:

            other = (
                conversation.members
                .exclude(user=request.user)
                .select_related("user__profile")
                .first()
            )

            other_user = other.user if other else None

        chats.append({
            "conversation": conversation,
            "name": conversation.display_name_for(request.user),
            "latest": latest,
            "unread": unread,
            "other_user": other_user,
        })

    return render(request, "messaging/inbox.html", {"chats": chats})

@login_required
@require_POST
def start_dm(request, user_id):

    other = get_object_or_404(User, pk=user_id, is_active=True)

    if other.pk == request.user.pk:
        return redirect("inbox")

    if is_blocked_between(request.user.pk, other.pk):

        messages.error(request, "You can't message this person.")

        return redirect("people")

    with transaction.atomic():

        candidates = (
            Conversation.objects
            .filter(is_group=False, members__user=request.user)
            .filter(members__user=other)
            .distinct()
        )

        conversation = None

        for candidate in candidates:
            if candidate.members.count() == 2:
                conversation = candidate
                break

        if conversation is None:

            conversation = Conversation.objects.create(
                is_group=False,
                created_by=request.user
            )

            Membership.objects.create(
                conversation=conversation,
                user=request.user
            )

            Membership.objects.create(
                conversation=conversation,
                user=other
            )

    return redirect("conversation", conversation_id=conversation.pk)

@login_required
def conversation_view(request, conversation_id):

    conversation = get_object_or_404(
        Conversation,
        pk=conversation_id,
        members__user=request.user
    )

    membership = get_object_or_404(
        Membership,
        conversation=conversation,
        user=request.user
    )

    if request.method == "POST":

        body = request.POST.get("body", "").strip()

        if body and len(body) <= 4000:

            Message.objects.create(
                conversation=conversation,
                sender=request.user,
                body=body
            )

        return redirect("conversation", conversation_id=conversation.pk)

    membership.last_read_at = timezone.now()
    membership.save(update_fields=["last_read_at"])

    blocked = blocked_ids_for(request.user)

    recent = list(
        conversation.messages
        .select_related("sender__profile")
        .exclude(sender_id__in=blocked)
        .order_by("-created_at")[:HISTORY_LIMIT]
    )

    recent.reverse()

    return render(
        request,
        "messaging/conversation.html",
        {
            "conversation": conversation,
            "conversation_name": conversation.display_name_for(request.user),
            "chat_messages": recent,
            "blocked_ids": sorted(blocked),
            "realtime": realtime.enabled(),
        }
    )


@login_required
def messages_since(request, conversation_id, last_id):
    """Messages newer than last_id. Used by the browser to catch up
    after the WebSocket reconnects."""

    get_object_or_404(
        Membership,
        conversation_id=conversation_id,
        user=request.user
    )

    msgs = (
        Message.objects
        .filter(conversation_id=conversation_id, id__gt=last_id)
        .exclude(sender_id__in=blocked_ids_for(request.user))
        .select_related("sender__profile")
        .order_by("id")[:HISTORY_LIMIT]
    )

    return JsonResponse({
        "messages": [
            {
                "id": m.id,
                "body": m.body,
                "sender": m.sender.username,
                "sender_id": m.sender_id,
                "created_at": m.created_at.isoformat(),
                **avatar_payload(m.sender),
            }
            for m in msgs
        ]
    })


@login_required
@require_POST
def send_message(request, conversation_id):
    """Save a message and broadcast it to the room.

    The browser sends messages here over plain HTTP so a send never
    silently disappears if the WebSocket happens to be reconnecting.
    The WebSocket is only used to receive.
    """

    get_object_or_404(
        Membership,
        conversation_id=conversation_id,
        user=request.user
    )

    conversation = get_object_or_404(Conversation, pk=conversation_id)

    if not conversation.is_group:

        other = (
            Membership.objects
            .filter(conversation=conversation)
            .exclude(user=request.user)
            .first()
        )

        if other and is_blocked_between(request.user.pk, other.user_id):
            return JsonResponse({"error": "blocked"}, status=403)

    try:
        data = json.loads(request.body)
        body = str(data.get("body", "")).strip()
    except (ValueError, AttributeError):
        return JsonResponse({"error": "bad request"}, status=400)

    if not body or len(body) > 4000:
        return JsonResponse({"error": "invalid message"}, status=400)

    # Basic flood protection: one message per half second per user.
    too_fast = Message.objects.filter(
        sender=request.user,
        created_at__gte=timezone.now() - timedelta(milliseconds=500)
    ).exists()

    if too_fast:
        return JsonResponse({"error": "slow down"}, status=429)

    message = Message.objects.create(
        conversation_id=conversation_id,
        sender=request.user,
        body=body
    )

    payload = {
        "id": message.id,
        "body": message.body,
        "sender": request.user.username,
        "sender_id": request.user.id,
        "created_at": message.created_at.isoformat(),
        **avatar_payload(request.user),
    }

    # Push to everyone in the room. If this fails the message is still
    # saved, and other browsers pick it up on their next catch-up.
    realtime.publish(conversation_id, payload)

    return JsonResponse(payload)

@login_required
@require_POST
def report_message(request, message_id):

    message = get_object_or_404(
        Message.objects.select_related("sender"),
        pk=message_id
    )

    # You can only report messages from rooms you are in.
    get_object_or_404(
        Membership,
        conversation_id=message.conversation_id,
        user=request.user
    )

    if message.sender_id == request.user.id:
        return JsonResponse({"error": "invalid"}, status=400)

    recent = Report.objects.filter(
        reporter=request.user,
        created_at__gte=timezone.now() - timedelta(hours=1)
    ).count()

    if recent >= REPORTS_PER_HOUR:
        return JsonResponse({"error": "slow down"}, status=429)

    try:
        reason = str(json.loads(request.body).get("reason", "")).strip()[:500]
    except (ValueError, AttributeError):
        reason = ""

    Report.objects.get_or_create(
        reporter=request.user,
        message=message,
        defaults={
            "reported_user": message.sender,
            "message_body": message.body,
            "reason": reason,
        }
    )

    return JsonResponse({"ok": True})


@login_required
def ably_auth(request):
    """Gives the browser a short-lived token to subscribe to one room."""

    conversation_id = request.GET.get("conversation", "")

    if not conversation_id.isdigit():
        return HttpResponse(status=400)

    get_object_or_404(
        Membership,
        conversation_id=int(conversation_id),
        user=request.user
    )

    token = realtime.make_subscribe_token(
        request.user.pk,
        int(conversation_id)
    )

    if not token:
        return HttpResponse(status=503)

    return HttpResponse(token, content_type="text/plain")

<script>

(() => {

    const conversationId = {{ conversation.id }};

    const box = document.getElementById("messages");
    const form = document.getElementById("message-form");
    const input = document.getElementById("message-input");
    const status = document.getElementById("status");

    const currentUserId = Number(box.dataset.userId);

    const realtimeEnabled = {{ realtime|yesno:"true,false" }};

    const seen = new Set();

    const blockedIds = new Set(
        JSON.parse(document.getElementById("blocked-ids").textContent)
    );

    let realtime = null;
    let statusTimer = null;
    let pollTimer = null;


    function formatTime(iso) {

        const d = new Date(iso);

        if (isNaN(d)) {
            return "";
        }

        return d.toLocaleTimeString([], {
            hour: "2-digit",
            minute: "2-digit"
        });

    }


    // Show server-rendered times in the viewer's own timezone.
    box.querySelectorAll("time[datetime]").forEach(t => {
        t.textContent = formatTime(t.getAttribute("datetime"));
    });

    box.querySelectorAll("[data-message-id]").forEach(el => {
        seen.add(Number(el.dataset.messageId));
    });


    function lastId() {

        let max = 0;

        seen.forEach(id => {
            if (id > max) {
                max = id;
            }
        });

        return max;

    }


    function scrollDown() {

        box.scrollTop = box.scrollHeight;

    }


    function addMessage(data) {

        const id = Number(data.id);

        if (seen.has(id)) {
            return;
        }

        seen.add(id);

        if (blockedIds.has(Number(data.sender_id))) {
            return;
        }


        const empty = document.getElementById("empty-state");

        if (empty) {
            empty.remove();
        }


        const mine = Number(data.sender_id) === currentUserId;

        const wrap = document.createElement("div");
        wrap.className = "message" + (mine ? " mine" : "");
        wrap.dataset.messageId = id;

        const meta = document.createElement("div");
        meta.className = "message-meta";

        const name = document.createElement("strong");
        name.textContent = mine ? "You" : data.sender;

        const time = document.createElement("time");
        time.setAttribute("datetime", data.created_at);
        time.textContent = formatTime(data.created_at);

        const avatarEl = document.createElement("div");
        avatarEl.className = "avatar";

        const color = /^#[0-9a-fA-F]{6}$/.test(data.sender_color || "")
            ? data.sender_color
            : "#4f46e5";

        avatarEl.style.cssText =
            `background:${color};color:#fff;width:20px;height:20px;font-size:11px`;

        avatarEl.textContent = data.sender_avatar || data.sender_initial || "?";

        meta.append(avatarEl, name, time);

        if (!mine) {

            const actions = document.createElement("span");
            actions.className = "msg-actions";

            const report = document.createElement("button");
            report.type = "button";
            report.className = "link-button";
            report.dataset.action = "report";
            report.dataset.messageId = id;
            report.textContent = "Report";

            const block = document.createElement("button");
            block.type = "button";
            block.className = "link-button";
            block.dataset.action = "block";
            block.dataset.userId = data.sender_id;
            block.textContent = "Block";

            actions.append(report, block);
            meta.append(actions);

        }

        const bubble = document.createElement("div");
        bubble.className = "bubble";
        bubble.textContent = data.body;

        wrap.append(meta, bubble);
        box.appendChild(wrap);

        scrollDown();

    }


    // After a reconnect, fetch anything we missed while disconnected.
    async function catchUp() {

        try {

            const res = await fetch(
                `/chat/${conversationId}/since/${lastId()}/`,
                { credentials: "same-origin" }
            );

            if (!res.ok) {
                return;
            }

            const data = await res.json();

            data.messages.forEach(addMessage);

        } catch (error) {

            console.error("Catch-up failed:", error);

        }

    }


    function startPolling() {

        clearInterval(pollTimer);

        pollTimer = setInterval(catchUp, 3000);

    }


    function connect() {

        // No realtime service available (or its script failed to load):
        // fall back to checking for new messages every few seconds.
        if (!realtimeEnabled || typeof Ably === "undefined") {

            catchUp();
            startPolling();

            return;

        }

        realtime = new Ably.Realtime({
            authUrl: "/ably-auth/",
            authParams: { conversation: String(conversationId) }
        });

        const channel = realtime.channels.get(`chat-${conversationId}`);

        channel.subscribe("message", msg => {

            try {

                addMessage(
                    typeof msg.data === "string"
                        ? JSON.parse(msg.data)
                        : msg.data
                );

            } catch (error) {

                console.error("Could not process message:", error);

            }

        });

        realtime.connection.on("connected", () => {

            clearTimeout(statusTimer);
            clearInterval(pollTimer);

            status.hidden = true;

            // Pick up anything that arrived while we were offline.
            catchUp();

        });

        const showReconnecting = () => {

            clearTimeout(statusTimer);

            statusTimer = setTimeout(() => {
                status.textContent = "Reconnecting…";
                status.hidden = false;
            }, 1500);

        };

        realtime.connection.on("disconnected", showReconnecting);
        realtime.connection.on("suspended", showReconnecting);

        realtime.connection.on("failed", () => {
            startPolling();
        });

        // Safety net: if we aren't connected after 10 seconds, poll instead.
        setTimeout(() => {

            if (realtime.connection.state !== "connected") {
                startPolling();
            }

        }, 10000);

    }


    document.addEventListener("visibilitychange", () => {

        if (!document.hidden) {
            catchUp();
        }

    });


    const csrf = form.querySelector("[name=csrfmiddlewaretoken]").value;

    let sending = false;
    let flashTimer = null;


    function flash(text) {

        clearTimeout(flashTimer);

        status.textContent = text;
        status.hidden = false;

        flashTimer = setTimeout(() => {
            status.textContent = "Reconnecting…";
            status.hidden = true;
        }, 3000);

    }


    // Messages are sent over HTTP (reliable); realtime only delivers them.
    form.addEventListener("submit", async event => {

        event.preventDefault();

        const body = input.value.trim();

        if (!body || sending) {
            return;
        }

        sending = true;

        try {

            const res = await fetch(`/chat/${conversationId}/send/`, {
                method: "POST",
                credentials: "same-origin",
                headers: {
                    "Content-Type": "application/json",
                    "X-CSRFToken": csrf
                },
                body: JSON.stringify({ body: body })
            });

            if (res.status === 429) {
                flash("Slow down a little.");
                return;
            }

            if (res.status === 403) {
                flash("You can't message this person.");
                return;
            }

            if (!res.ok) {
                throw new Error("HTTP " + res.status);
            }

            addMessage(await res.json());

            input.value = "";

        } catch (error) {

            console.error("Send failed:", error);

            flash("Couldn't send. Your message is still in the box, try again.");

        } finally {

            sending = false;
            input.focus();

        }

    });


    function postJson(url, payload) {

        return fetch(url, {
            method: "POST",
            credentials: "same-origin",
            headers: {
                "Content-Type": "application/json",
                "X-CSRFToken": csrf
            },
            body: JSON.stringify(payload)
        });

    }


    // Report / Block buttons (works for old and newly arrived messages).
    box.addEventListener("click", async event => {

        const button = event.target.closest("[data-action]");

        if (!button) {
            return;
        }

        try {

            if (button.dataset.action === "report") {

                const reason = window.prompt(
                    "Why are you reporting this message? (optional)"
                );

                if (reason === null) {
                    return;
                }

                const res = await postJson(
                    `/messages/${button.dataset.messageId}/report/`,
                    { reason: reason }
                );

                flash(
                    res.ok
                        ? "Thanks, we'll take a look."
                        : "Couldn't send the report."
                );

            }

            if (button.dataset.action === "block") {

                const ok = window.confirm(
                    "Block this person? You won't see their messages " +
                    "and they won't be able to message you."
                );

                if (!ok) {
                    return;
                }

                const res = await postJson(
                    `/people/${button.dataset.userId}/block/`,
                    {}
                );

                if (res.ok) {
                    window.location.reload();
                } else {
                    flash("Couldn't block that person.");
                }

            }

        } catch (error) {

            console.error(error);

            flash("Something went wrong.");

        }

    });


    window.addEventListener("beforeunload", () => {

        if (realtime) {
            realtime.close();
        }

    });


    scrollDown();

    connect();

})();

</script>

{% endblock %}
